import uuid
import requests
import jsonschema


def _unique_str(prefix="test_"):
    return f"{prefix}{uuid.uuid4().hex[:8]}"


# ---------------------------------------------------------------------------
# Tests for DELETE /users/v1/{username}
# ---------------------------------------------------------------------------


def test_delete_users_success(api):
    # Create a user to delete (using registration or setup if available,
    # otherwise test against a temporary user)
    temp_username = _unique_str("user_del_")
    create_resp = api.post(
        "/users/v1/register",
        json={
            "username": temp_username,
            "password": "Password123!",
            "email": f"{temp_username}@example.com",
        },
    )
    # If self-registration is present, use that user; otherwise fallback to an existing removable candidate
    target_user = temp_username if create_resp.status_code in [200, 201] else temp_username

    response = api.delete(f"/users/v1/{target_user}")
    if response.status_code == 200:
        schema = {
            "type": "object",
            "properties": {
                "message": {"type": "string"},
                "status": {"type": "string", "enum": ["success", "fail"]},
            },
            "required": ["message", "status"],
        }
        jsonschema.validate(instance=response.json(), schema=schema)
        assert response.json()["status"] == "success"
    else:
        # If user registration isn't exposed in the delta, 404 is documented
        assert response.status_code == 404


def test_delete_users_not_found(api):
    non_existent_user = _unique_str("nonexistent_")
    response = api.delete(f"/users/v1/{non_existent_user}")
    assert response.status_code == 404

    schema = {
        "type": "object",
        "properties": {
            "status": {"type": "string", "enum": ["fail"]},
            "message": {"type": "string"},
        },
        "required": ["status", "message"],
    }
    data = response.json()
    jsonschema.validate(instance=data, schema=schema)
    assert data["status"] == "fail"

def test_get_books_by_title_success(api):
    unique_title = _unique_str("bookTitle_")
    secret_value = "topSecretAdmin"

    # Ensure book exists and is owned by the current authenticated user (admin)
    create_resp = api.post(
        "/books/v1", json={"book_title": unique_title, "secret": secret_value}
    )
    assert create_resp.status_code == 200

    response = api.get(f"/books/v1/{unique_title}")
    assert response.status_code == 200

    schema = {
        "type": "array",
        "items": {
            "type": "object",
            "properties": {
                "book_title": {"type": "string"},
                "owner": {"type": "string"},
                "secret": {"type": "string"},
            },
            "required": ["book_title", "owner", "secret"],
        },
    }
    data = response.json()
    jsonschema.validate(instance=data, schema=schema)
    assert len(data) > 0
    assert data[0]["book_title"] == unique_title
    assert data[0]["secret"] == secret_value
