```python
import uuid
import pytest
import requests
from jsonschema import validate

USERS_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["data", "count"],
    "properties": {
        "count": {"type": "integer"},
        "data": {
            "type": "array",
            "items": {
                "type": "object",
                "required": ["email", "id"],
                "properties": {
                    "email": {"type": "string", "format": "email"},
                    "is_active": {"type": "boolean"},
                    "is_superuser": {"type": "boolean"},
                    "full_name": {"type": ["string", "null"]},
                    "id": {"type": "string"},
                    "created_at": {"type": ["string", "null"]},
                },
            },
        },
    },
}

ITEM_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["title", "id", "owner_id"],
    "properties": {
        "title": {"type": "string", "minLength": 1, "maxLength": 255},
        "description": {"type": ["string", "null"], "maxLength": 255},
        "id": {"type": "string"},
        "owner_id": {"type": "string"},
        "created_at": {"type": ["string", "null"]},
    },
}

MESSAGE_SCHEMA = {
    "type": "object",
    "required": ["message"],
    "properties": {
        "message": {"type": "string"},
    },
}

HTTP_VALIDATION_ERROR_SCHEMA = {
    "type": "object",
    "required": ["detail"],
    "properties": {
        "detail": {
            "type": "array",
            "items": {
                "type": "object",
                "required": ["loc", "msg", "type"],
                "properties": {
                    "loc": {
                        "type": "array",
                        "items": {"type": ["string", "integer"]},
                    },
                    "msg": {"type": "string"},
                    "type": {"type": "string"},
                    "input": {},
                    "ctx": {"type": "object"},
                },
            },
        }
    },
}


def test_get_users_default_pagination(api):
    response = api.get("/api/v1/users/")
    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=USERS_PUBLIC_SCHEMA)
    assert isinstance(data["data"], list)
    assert data["count"] >= 0


def test_get_users_custom_pagination(api):
    response = api.get("/api/v1/users/", params={"skip": 1, "limit": 2})
    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=USERS_PUBLIC_SCHEMA)
    assert len(data["data"]) <= 2


def test_get_users_invalid_pagination_type(api):
    response = api.get("/api/v1/users/", params={"skip": "invalid", "limit": 10})
    assert response.status_code == 422
    data = response.json()
    validate(instance=data, schema=HTTP_VALIDATION_ERROR_SCHEMA)


def test_patch_users_me_password_invalid_short_password(api):
    payload = {
        "current_password": "short",
        "new_password": "short",
    }
    response = api.patch("/api/v1/users/me/password", json=payload)
    assert response.status_code == 422
    data = response.json()
    validate(instance=data, schema=HTTP_VALIDATION_ERROR_SCHEMA)


def test_patch_users_me_password_missing_fields(api):

    response = api.patch("/api/v1/users/me/password", json=payload)
    assert response.status_code == 422

def test_post_items_success(api):
    item_payload = {
        "title": f"Test Item {uuid.uuid4().hex[:8]}",
        "description": "A description for the test item",
    }
    response = api.post("/api/v1/items/", json=item_payload)
    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=ITEM_PUBLIC_SCHEMA)
    assert data["title"] == item_payload["title"]
    assert data["description"] == item_payload["description"]

    # Cleanup
    api.delete(f"/api/v1/items/{data['id']}")


def test_post_items_missing_required_title(api):

    response = api.post("/api/v1/items/", json=item_payload)
    assert response.status_code == 422
    data = response.json()
    validate(instance=data, schema=HTTP_VALIDATION_ERROR_SCHEMA)


def test_post_items_empty_title(api):

def test_put_items_success(api):
    create_payload = {"title": f"Initial Title {uuid.uuid4().hex[:8]}"}
    create_res = api.post("/api/v1/items/", json=create_payload)
    assert create_res.status_code == 200
    item_id = create_res.json()["id"]

    try:
        update_payload = {
            "title": f"Updated Title {uuid.uuid4().hex[:8]}",
            "description": "Updated description",
        }
        update_res = api.put(f"/api/v1/items/{item_id}", json=update_payload)
        assert update_res.status_code == 200
        data = update_res.json()
        validate(instance=data, schema=ITEM_PUBLIC_SCHEMA)
        assert data["title"] == update_payload["title"]
        assert data["description"] == update_payload["description"]
        assert data["id"] == item_id
    finally:
        api.delete(f"/api/v1/items/{item_id}")


def test_put_items_invalid_uuid(api):
    update_payload = {"title": "Updated Title"}
    response = api.put("/api/v1/items/not-a-valid-uuid", json=update_payload)
    assert response.status_code == 422
    data = response.json()
    validate(instance=data, schema=HTTP_VALIDATION_ERROR_SCHEMA)

    item_id = create_res.json()["id"]

    finally:
        api.delete(f"/api/v1/items/{item_id}")


def test_delete_items_success(api):
    create_payload = {"title": f"To Be Deleted {uuid.uuid4().hex[:8]}"}
    create_res = api.post("/api/v1/items/", json=create_payload)
    assert create_res.status_code == 200
    item_id = create_res.json()["id"]

    delete_res = api.delete(f"/api/v1/items/{item_id}")
    assert delete_res.status_code == 200
    data = delete_res.json()
    validate(instance=data, schema=MESSAGE_SCHEMA)
    assert isinstance(data["message"], str)
```
