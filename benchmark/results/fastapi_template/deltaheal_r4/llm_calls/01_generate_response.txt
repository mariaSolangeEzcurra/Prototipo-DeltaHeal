```python
import uuid
import pytest
import requests
from jsonschema import validate

# JSON Schemas basades en el contrato OpenAPI
USER_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["email", "id"],
    "properties": {
        "email": {"type": "string", "maxLength": 255},
        "is_active": {"type": "boolean"},
        "is_superuser": {"type": "boolean"},
        "full_name": {"type": ["string", "null"], "maxLength": 255},
        "id": {"type": "string"},
        "created_at": {"type": ["string", "null"]},
    },
}

USERS_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["data", "count"],
    "properties": {
        "data": {
            "type": "array",
            "items": USER_PUBLIC_SCHEMA,
        },
        "count": {"type": "integer"},
    },
}

ITEM_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["title", "id", "owner_id"],
    "properties": {
        "title": {"type": "string", "minLength": 1, "maxLength": 255},
        "description": {"type": ["string", "null"], "maxLength": 255},
        "id": {"type": "string"},
        "owner_id": {"type": "string"},
        "created_at": {"type": ["string", "null"]},
    },
}

MESSAGE_SCHEMA = {
    "type": "object",
    "required": ["message"],
    "properties": {
        "message": {"type": "string"},
    },
}

HTTP_VALIDATION_ERROR_SCHEMA = {
    "type": "object",
    "required": ["detail"],
    "properties": {
        "detail": {
            "type": "array",
            "items": {
                "type": "object",
                "required": ["loc", "msg", "type"],
                "properties": {
                    "loc": {
                        "type": "array",
                        "items": {"type": ["string", "integer"]},
                    },
                    "msg": {"type": "string"},
                    "type": {"type": "string"},
                    "input": {},
                    "ctx": {"type": "object"},
                },
            },
        }
    },
}


# --- Operación: GET /api/v1/users/ ---

def test_get_users_default_success(api):
    response = api.get("/api/v1/users/")
    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=USERS_PUBLIC_SCHEMA)
    assert isinstance(data["data"], list)
    assert isinstance(data["count"], int)
    assert data["count"] >= 0


def test_get_users_with_pagination(api):
    response = api.get("/api/v1/users/", params={"skip": 0, "limit": 5})
    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=USERS_PUBLIC_SCHEMA)
    assert len(data["data"]) <= 5


def test_get_users_invalid_query_params(api):
    response = api.get("/api/v1/users/", params={"skip": "invalid_integer"})
    assert response.status_code == 422
    validate(instance=response.json(), schema=HTTP_VALIDATION_ERROR_SCHEMA)


# --- Operación: PATCH /api/v1/users/me/password ---

def test_patch_users_me_password_validation_error_length(api):
    # La contraseña debe tener al menos 8 caracteres según el contrato
    payload = {
        "current_password": "short",
        "new_password": "short"
    }
    response = api.patch("/api/v1/users/me/password", json=payload)
    assert response.status_code == 422
    validate(instance=response.json(), schema=HTTP_VALIDATION_ERROR_SCHEMA)


def test_patch_users_me_password_validation_error_missing_fields(api):
    payload = {
        "current_password": "changethis"
    }
    response = api.patch("/api/v1/users/me/password", json=payload)
    assert response.status_code == 422
    validate(instance=response.json(), schema=HTTP_VALIDATION_ERROR_SCHEMA)


def test_patch_users_me_password_roundtrip_success(api):
    # Cambiar a una nueva contraseña válida y restablecer de inmediato para no romper el usuario admin
    original_pass = "changethis"
    temp_pass = "newpassword123"

    # 1. Cambiar a temp_pass
    update_res = api.patch(
        "/api/v1/users/me/password",
        json={"current_password": original_pass, "new_password": temp_pass}
    )
    assert update_res.status_code == 200
    validate(instance=update_res.json(), schema=MESSAGE_SCHEMA)

    # 2. Revertir inmediatamente a original_pass
    revert_res = api.patch(
        "/api/v1/users/me/password",
        json={"current_password": temp_pass, "new_password": original_pass}
    )
    assert revert_res.status_code == 200
    validate(instance=revert_res.json(), schema=MESSAGE_SCHEMA)


# --- Operación: POST /api/v1/items/ ---

def test_post_items_success_minimal(api):
    payload = {
        "title": f"Item-{uuid.uuid4().hex[:8]}"
    }
    response = api.post("/api/v1/items/", json=payload)
    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=ITEM_PUBLIC_SCHEMA)
    assert data["title"] == payload["title"]
    assert data["description"] is None

    # Limpieza
    api.delete(f"/api/v1/items/{data['id']}")


def test_post_items_success_full(api):
    payload = {
        "title": f"Item-{uuid.uuid4().hex[:8]}",
        "description": "Item with description"
    }
    response = api.post("/api/v1/items/", json=payload)
    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=ITEM_PUBLIC_SCHEMA)
    assert data["title"] == payload["title"]
    assert data["description"] == payload["description"]

    # Limpieza
    api.delete(f"/api/v1/items/{data['id']}")

    }

def test_post_items_validation_error_empty_title(api):

# --- Operación: PUT /api/v1/items/{id} ---

def test_put_item_success(api):
    # Crear item primero
    create_res = api.post(
        "/api/v1/items/",
        json={"title": f"Orig-{uuid.uuid4().hex[:8]}", "description": "Original Desc"}
    )
    assert create_res.status_code == 200
    item_id = create_res.json()["id"]

    # Actualizar item
    update_payload = {
        "title": f"Updated-{uuid.uuid4().hex[:8]}",
        "description": "Updated Description"
    }
    update_res = api.put(f"/api/v1/items/{item_id}", json=update_payload)
    assert update_res.status_code == 200
    updated_data = update_res.json()
    validate(instance=updated_data, schema=ITEM_PUBLIC_SCHEMA)
    assert updated_data["title"] == update_payload["title"]
    assert updated_data["description"] == update_payload["description"]

    # Limpieza
    api.delete(f"/api/v1/items/{item_id}")


def test_put_item_validation_error_invalid_uuid(api):
    response = api.put("/api/v1/items/not-a-valid-uuid", json={"title": "Updated Title"})
    assert response.status_code == 422
    validate(instance=response.json(), schema=HTTP_VALIDATION_ERROR_SCHEMA)


def test_put_item_validation_error_empty_title(api):
    # Generar un UUID válido para validar fallo en el requestBody
    valid_uuid = str(uuid.uuid4())
    response = api.put(f"/api/v1/items/{valid_uuid}", json={"title": ""})
    assert response.status_code == 422
    validate(instance=response.json(), schema=HTTP_VALIDATION_ERROR_SCHEMA)


# --- Operación: DELETE /api/v1/items/{id} ---

def test_delete_item_success(api):
    # Crear item
    create_res = api.post(

    del_res = api.delete(f"/api/v1/items/{item_id}")
```
