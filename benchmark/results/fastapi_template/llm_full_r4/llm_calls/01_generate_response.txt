import uuid

import requests
from jsonschema import validate

# JSON Schemas for response validation based on OpenAPI contract
SCHEMA_TOKEN = {
    "type": "object",
    "required": ["access_token"],
    "properties": {
        "access_token": {"type": "string"},
        "token_type": {"type": "string"},
    },
}

SCHEMA_USER_PUBLIC = {
    "type": "object",
    "required": ["email", "id"],
    "properties": {
        "email": {"type": "string"},
        "is_active": {"type": "boolean"},
        "is_superuser": {"type": "boolean"},
        "full_name": {"type": ["string", "null"]},
        "id": {"type": "string", "format": "uuid"},
        "created_at": {"type": ["string", "null"]},
    },
}

SCHEMA_USERS_PUBLIC = {
    "type": "object",
    "required": ["data", "count"],
    "properties": {
        "data": {"type": "array", "items": SCHEMA_USER_PUBLIC},
        "count": {"type": "integer"},
    },
}

SCHEMA_ITEM_PUBLIC = {
    "type": "object",
    "required": ["title", "id", "owner_id"],
    "properties": {
        "title": {"type": "string"},
        "description": {"type": ["string", "null"]},
        "id": {"type": "string", "format": "uuid"},
        "owner_id": {"type": "string", "format": "uuid"},
        "created_at": {"type": ["string", "null"]},
    },
}

SCHEMA_ITEMS_PUBLIC = {
    "type": "object",
    "required": ["data", "count"],
    "properties": {
        "data": {"type": "array", "items": SCHEMA_ITEM_PUBLIC},
        "count": {"type": "integer"},
    },
}

SCHEMA_MESSAGE = {
    "type": "object",
    "required": ["message"],
    "properties": {
        "message": {"type": "string"},
    },
}

SCHEMA_VALIDATION_ERROR = {
    "type": "object",
    "required": ["detail"],
    "properties": {
        "detail": {
            "type": "array",
            "items": {
                "type": "object",
                "required": ["loc", "msg", "type"],
                "properties": {
                    "loc": {"type": "array"},
                    "msg": {"type": "string"},
                    "type": {"type": "string"},
                },
            },
        }
    },
}

def test_delete_user_me_as_separate_user(base_url):
    # Create a non-admin dedicated user and delete its own account
    unique_str = uuid.uuid4().hex[:8]
    email = f"selfdelete_{unique_str}@example.com"
    password = "SelfDeletePassword123!"

    reg_resp = requests.post(
        f"{base_url}/api/v1/users/signup",
        json={"email": email, "password": password, "full_name": "Self Delete"}
    )
    assert reg_resp.status_code == 200

    token_resp = requests.post(
        f"{base_url}/api/v1/login/access-token",
        data={"username": email, "password": password}
    )
    assert token_resp.status_code == 200
    access_token = token_resp.json()["access_token"]

    user_session = requests.Session()
    user_session.headers.update({"Authorization": f"Bearer {access_token}"})

    response = user_session.delete(f"{base_url}/api/v1/users/me")
    assert response.status_code == 200
    validate(instance=response.json(), schema=SCHEMA_MESSAGE)
