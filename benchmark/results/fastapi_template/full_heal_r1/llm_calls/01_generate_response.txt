# JSON Schemas from OpenAPI Contract
USER_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["email", "id"],
    "properties": {
        "email": {"type": "string"},
        "is_active": {"type": "boolean"},
        "is_superuser": {"type": "boolean"},
        "full_name": {"type": ["string", "null"]},
        "id": {"type": "string", "format": "uuid"},
        "created_at": {"type": ["string", "null"]},
    },
}

USERS_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["data", "count"],
    "properties": {
        "data": {"type": "array", "items": USER_PUBLIC_SCHEMA},
        "count": {"type": "integer"},
    },
}

TOKEN_SCHEMA = {
    "type": "object",
    "required": ["access_token"],
    "properties": {
        "access_token": {"type": "string"},
        "token_type": {"type": "string"},
    },
}

MESSAGE_SCHEMA = {
    "type": "object",
    "required": ["message"],
    "properties": {
        "message": {"type": "string"},
    },
}

ITEM_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["title", "id", "owner_id"],
    "properties": {
        "title": {"type": "string"},
        "description": {"type": ["string", "null"]},
        "id": {"type": "string", "format": "uuid"},
        "owner_id": {"type": "string", "format": "uuid"},
        "created_at": {"type": ["string", "null"]},
    },
}

ITEMS_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["data", "count"],
    "properties": {
        "data": {"type": "array", "items": ITEM_PUBLIC_SCHEMA},
        "count": {"type": "integer"},
    },
}

HTTP_VALIDATION_ERROR_SCHEMA = {
    "type": "object",
    "properties": {
        "detail": {
            "type": "array",
            "items": {
                "type": "object",
                "required": ["loc", "msg", "type"],
                "properties": {
                    "loc": {"type": "array"},
                    "msg": {"type": "string"},
                    "type": {"type": "string"},
                },
            },
        }
    },
}

def random_email():
    return f"test_{uuid.uuid4().hex[:10]}@example.com"


# --- Login / Auth Tests ---
def test_post_login_access_token_success(api):
    payload = {"username": "admin@example.com", "password": "changethis"}
    response = api.post("/api/v1/login/access-token", data=payload)
    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=TOKEN_SCHEMA)
    assert "access_token" in data
    assert data.get("token_type") == "bearer"


def test_post_login_access_token_validation_error(api):

    response = api.post("/api/v1/login/access-token", data={"username": "admin@example.com"})

    validate(instance=response.json(), schema=HTTP_VALIDATION_ERROR_SCHEMA)

def test_delete_user_me_flow(base_url):
    # Register a new user, log in as that user, and delete own account
    email = random_email()
    password = "delete_me_pass123"
    reg_payload = {"email": email, "password": password, "full_name": "Ephemeral User"}
    reg_resp = requests.post(f"{base_url}/api/v1/users/signup", json=reg_payload)
    assert reg_resp.status_code == 200

    login_resp = requests.post(
        f"{base_url}/api/v1/login/access-token",
        data={"username": email, "password": password},
    )
    assert login_resp.status_code == 200
    token = login_resp.json()["access_token"]

    user_session = requests.Session()
    user_session.headers.update({"Authorization": f"Bearer {token}"})

    del_resp = user_session.delete(f"{base_url}/api/v1/users/me")
    assert del_resp.status_code == 200
    validate(instance=del_resp.json(), schema=MESSAGE_SCHEMA)
