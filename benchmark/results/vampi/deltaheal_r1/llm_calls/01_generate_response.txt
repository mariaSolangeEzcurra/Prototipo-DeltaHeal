import uuid
import requests
import jsonschema


def generate_unique_string(prefix="test_"):
    return f"{prefix}{uuid.uuid4().hex[:8]}"


# ---------------------------------------------------------------------------
# Operación: DELETE /users/v1/{username}
# ---------------------------------------------------------------------------


def test_delete_user_success(api):
    # Se crea un usuario temporal primero para mantener la prueba independiente y reproducible
    username = generate_unique_string("user_")
    password = "password123"
    create_payload = {
        "username": username,
        "password": password,
        "email": f"{username}@example.com",
    }
    create_res = api.post("/users/v1/register", json=create_payload)
    # Si el endpoint de registro no es estándar, se asume que al menos el usuario se crea o existe
    if create_res.status_code not in (200, 201):
        # Fallback a un intento de borrado directo si ya existiera, pero idealmente se creó
        pass

    res = api.delete(f"/users/v1/{username}")
    assert res.status_code == 200

    schema = {
        "type": "object",
        "properties": {
            "message": {"type": "string"},
            "status": {"type": "string", "enum": ["success", "fail"]},
        },
        "required": ["message", "status"],
    }
    jsonschema.validate(instance=res.json(), schema=schema)
    assert res.json()["status"] == "success"


def test_delete_user_not_found(api):
    non_existent_username = f"non_existent_{uuid.uuid4().hex}"
    res = api.delete(f"/users/v1/{non_existent_username}")
    assert res.status_code == 404

    schema = {
        "type": "object",
        "properties": {
            "status": {"type": "string", "enum": ["fail"]},

def test_get_book_by_title_success(api):
    title = generate_unique_string("bookTitle_")
    secret = "owner_classified_info"
    post_res = api.post("/books/v1", json={"book_title": title, "secret": secret})
    assert post_res.status_code == 200

    res = api.get(f"/books/v1/{title}")
    assert res.status_code == 200

    schema = {
        "type": "array",
        "items": {
            "type": "object",
            "properties": {
                "book_title": {"type": "string"},
                "owner": {"type": "string"},
                "secret": {"type": "string"},
            },
            "required": ["book_title", "owner", "secret"],
        },
    }
    data = res.json()
    jsonschema.validate(instance=data, schema=schema)
    assert len(data) > 0
    assert data[0]["book_title"] == title
    assert data[0]["secret"] == secret
