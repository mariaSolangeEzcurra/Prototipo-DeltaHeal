# Helper functions
def generate_unique_str(prefix="test_"):
    return f"{prefix}{uuid.uuid4().hex[:8]}"

# ---------------------------------------------------------

    assert response.status_code == 200

    schema = {
        "type": "object",
        "properties": {
            "message": {"type": "string"}
        }
    }
    validate(instance=response.json(), schema=schema)
