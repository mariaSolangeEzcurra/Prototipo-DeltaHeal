import uuid
import requests
import jsonschema


def _unique_str(prefix="test_"):
    return f"{prefix}{uuid.uuid4().hex[:8]}"


# ---------------------------------------------------------------------------
# Tests for DELETE /users/v1/{username}
# ---------------------------------------------------------------------------

def test_delete_user_success(api):
    # Register a new user first using standard VAmPI endpoint to avoid breaking existing users
    username = _unique_str("user_del_")
    reg_payload = {
        "username": username,
        "password": "Password123!",
        "email": f"{username}@example.com"
    }
    # Attempt registration if available
    api.post("/users/v1/register", json=reg_payload)

    # Delete the user as admin
    response = api.delete(f"/users/v1/{username}")
    assert response.status_code == 200

    schema = {
        "type": "object",
        "properties": {
            "message": {"type": "string"},
            "status": {"type": "string", "enum": ["success", "fail"]}
        },
        "required": ["message", "status"]
    }
    data = response.json()
    jsonschema.validate(instance=data, schema=schema)
    assert data["status"] == "success"


def test_delete_user_not_found(api):
    non_existent_user = _unique_str("nonexistent_")
    response = api.delete(f"/users/v1/{non_existent_user}")

    schema = {
        "type": "object",

def test_put_user_password_success(api):
    # Create a user to change password safely without corrupting seeded accounts
    username = _unique_str("pwd_user_")
    reg_payload = {
        "username": username,
        "password": "initialPassword123!",
        "email": f"{username}@example.com"
    }
    api.post("/users/v1/register", json=reg_payload)

    update_payload = {"password": "newSecurePassword456!"}
    response = api.put(f"/users/v1/{username}/password", json=update_payload)
    assert response.status_code == 204
    assert response.text == ""

def test_get_book_by_title_success(api):
    book_title = _unique_str("bookGet_")
    secret = "ownerSecret123"
    create_payload = {
        "book_title": book_title,
        "secret": secret
    }
    # Create the book with current session user (admin)
    create_resp = api.post("/books/v1", json=create_payload)
    assert create_resp.status_code == 200

    # Retrieve the book as the owner
    response = api.get(f"/books/v1/{book_title}")
    assert response.status_code == 200

    schema = {
        "type": "array",
        "items": {
            "type": "object",
            "properties": {
                "book_title": {"type": "string"},
                "owner": {"type": "string"},
                "secret": {"type": "string"}
            },
            "required": ["book_title", "owner", "secret"]
        }
    }
    data = response.json()
    jsonschema.validate(instance=data, schema=schema)
    assert len(data) > 0
    assert data[0]["book_title"] == book_title
    assert data[0]["secret"] == secret
