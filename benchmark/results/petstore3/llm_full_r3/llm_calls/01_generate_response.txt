import uuid
import random
import string
import pytest
from jsonschema import validate

# ---------------------------------------------------------------------------
# Schemas for validation derived from OpenAPI specification
# ---------------------------------------------------------------------------
CATEGORY_SCHEMA = {
    "type": "object",
    "properties": {
        "id": {"type": "integer"},
        "name": {"type": "string"},
    },
}

TAG_SCHEMA = {
    "type": "object",
    "properties": {
        "id": {"type": "integer"},
        "name": {"type": "string"},
    },
}

PET_SCHEMA = {
    "type": "object",
    "required": ["name", "photoUrls"],
    "properties": {
        "id": {"type": "integer"},
        "name": {"type": "string"},
        "category": CATEGORY_SCHEMA,
        "photoUrls": {"type": "array", "items": {"type": "string"}},
        "tags": {"type": "array", "items": TAG_SCHEMA},
        "status": {"type": "string", "enum": ["available", "pending", "sold"]},
    },
}

ORDER_SCHEMA = {
    "type": "object",
    "properties": {
        "id": {"type": "integer"},
        "petId": {"type": "integer"},
        "quantity": {"type": "integer"},
        "shipDate": {"type": "string"},
        "status": {"type": "string", "enum": ["placed", "approved", "delivered"]},
        "complete": {"type": "boolean"},
    },
}

USER_SCHEMA = {
    "type": "object",
    "properties": {
        "id": {"type": "integer"},
        "username": {"type": "string"},
        "firstName": {"type": "string"},
        "lastName": {"type": "string"},
        "email": {"type": "string"},
        "password": {"type": "string"},
        "phone": {"type": "string"},
        "userStatus": {"type": "integer"},
    },
}

API_RESPONSE_SCHEMA = {
    "type": "object",
    "properties": {
        "code": {"type": "integer"},
        "type": {"type": "string"},
        "message": {"type": "string"},
    },
}


# ---------------------------------------------------------------------------
# Helpers
# ---------------------------------------------------------------------------
def _random_str(length=8):
    return "".join(random.choices(string.ascii_lowercase, k=length))


def _generate_pet_payload(status="available", tag_name=None):
    pet_id = random.randint(100000, 999999)
    tags = [{"id": 1, "name": tag_name}] if tag_name else []
    return {
        "id": pet_id,
        "name": f"pet_{_random_str()}",
        "category": {"id": 1, "name": "Dogs"},
        "photoUrls": ["https://example.com/photo1.jpg"],
        "tags": tags,
        "status": status,
    }


def _generate_user_payload():
    return {
        "id": random.randint(10000, 99999),
        "username": f"user_{_random_str(10)}",
        "firstName": "John",
        "lastName": "Doe",
        "email": f"{_random_str()}@example.com",
        "password": "secretPassword123",
        "phone": "1234567890",
        "userStatus": 1,
    }


# ---------------------------------------------------------------------------
# Tests: /pet (POST & PUT)
# ---------------------------------------------------------------------------
def test_post_pet_success(api):
    payload = _generate_pet_payload()
    res = api.post("/pet", json=payload)
    assert res.status_code == 200
    data = res.json()
    validate(instance=data, schema=PET_SCHEMA)
    assert data["name"] == payload["name"]


def test_post_pet_invalid_input(api):
    # Invalid: missing required fields 'name' and 'photoUrls'
    res = api.post("/pet", json={"status": "invalid_status_value"})
    assert res.status_code in (400, 422)


def test_put_pet_success(api):
    pet_payload = _generate_pet_payload()
    create_res = api.post("/pet", json=pet_payload)
    assert create_res.status_code == 200
    created_pet = create_res.json()

    updated_payload = {
        "id": created_pet.get("id", pet_payload["id"]),
        "name": f"updated_{_random_str()}",
        "photoUrls": ["https://example.com/updated.jpg"],
        "status": "sold",
    }
    update_res = api.put("/pet", json=updated_payload)
    assert update_res.status_code == 200
    data = update_res.json()
    validate(instance=data, schema=PET_SCHEMA)
    assert data["name"] == updated_payload["name"]
    assert data["status"] == "sold"


def test_put_pet_invalid_id(api):
