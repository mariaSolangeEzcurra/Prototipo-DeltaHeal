import random
import uuid
import pytest
from jsonschema import validate

# ---------------------------------------------------------------------------
# Schemas JSON Schema según componentes del contrato OpenAPI
# ---------------------------------------------------------------------------

TAG_SCHEMA = {
    "type": "object",
    "properties": {
        "id": {"type": "integer"},
        "name": {"type": "string"},
    },
}

CATEGORY_SCHEMA = {
    "type": "object",
    "properties": {
        "id": {"type": "integer"},
        "name": {"type": "string"},
    },
}

PET_SCHEMA = {
    "type": "object",
    "required": ["name", "photoUrls"],
    "properties": {
        "id": {"type": "integer"},
        "name": {"type": "string"},
        "category": CATEGORY_SCHEMA,
        "photoUrls": {
            "type": "array",
            "items": {"type": "string"},
        },
        "tags": {
            "type": "array",
            "items": TAG_SCHEMA,
        },
        "status": {
            "type": "string",
            "enum": ["available", "pending", "sold"],
        },
    },
}

PET_LIST_SCHEMA = {
    "type": "array",
    "items": PET_SCHEMA,
}


def _generar_payload_mascota(pet_id=None, name=None, status="available", tag_name=None):
    """Genera un diccionario válido que conforma el esquema Pet."""
    pet_id = pet_id if pet_id is not None else random.randint(100000, 999999)
    tag_id = random.randint(1, 1000)
    tag_name = tag_name or f"tag-{uuid.uuid4().hex[:6]}"
    return {
        "id": pet_id,
        "name": name or f"pet-{uuid.uuid4().hex[:6]}",
        "category": {"id": 1, "name": "Dogs"},
        "photoUrls": [f"https://example.com/photos/{pet_id}.jpg"],
        "tags": [{"id": tag_id, "name": tag_name}],
        "status": status,
    }


# ===========================================================================
# Pruebas POST /pet (addPet)
# ===========================================================================

def test_post_pet_creacion_exitosa_200(api):
    """Verifica que POST /pet crea una mascota correctamente (200)."""
    payload = _generar_payload_mascota()
    response = api.post("/pet", json=payload)

    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=PET_SCHEMA)
    assert data["name"] == payload["name"]
    assert data["photoUrls"] == payload["photoUrls"]


def test_post_pet_faltan_campos_requeridos_error(api):
    """Verifica error (400 o 422) cuando se envían datos sin los campos requeridos."""
    # name y photoUrls son requeridos según el contrato
    payload_invalido = {"id": random.randint(100000, 999999)}
    response = api.post("/pet", json=payload_invalido)

    assert response.status_code in (400, 422)


def test_post_pet_cuerpo_invalido_formato_400(api):
    """Verifica 400 o 422 al enviar un cuerpo con tipo de dato incompatible."""
    response = api.post("/pet", json="esto_no_es_un_objeto_json")
    assert response.status_code in (400, 422)


# ===========================================================================
# Pruebas PUT /pet (updatePet)
# ===========================================================================

def test_put_pet_actualizacion_exitosa_200(api):
    """Verifica que PUT /pet actualiza una mascota existente (200)."""
    payload_creacion = _generar_payload_mascota()
    resp_creacion = api.post("/pet", json=payload_creacion)
    assert resp_creacion.status_code == 200
    pet_creada = resp_creacion.json()

    payload_actualizado = {
        **payload_creacion,
        "id": pet_creada["id"],
        "name": f"actualizado-{uuid.uuid4().hex[:6]}",
        "status": "sold",
    }
    response = api.put("/pet", json=payload_actualizado)

    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=PET_SCHEMA)
    assert data["name"] == payload_actualizado["name"]
    assert data["status"] == "sold"


def test_put_pet_no_encontrada_404(api):
    """Verifica 404 o 400 cuando se intenta actualizar una mascota inexistente."""
    id_inexistente = 999999999999
    payload = _generar_payload_mascota(pet_id=id_inexistente)
    response = api.put("/pet", json=payload)

    # 404 o 400 son documentados para pet no encontrada / id inválido
    assert response.status_code in (400, 404)


def test_put_pet_cuerpo_invalido_422_o_400(api):
    """Verifica 400 o 422 al enviar cuerpo inválido en PUT /pet."""
    response = api.put("/pet", json={"status": "invalido_sin_campos_obligatorios"})
    assert response.status_code in (400, 422)

def test_get_pet_por_id_exitoso_200(api):
    """Crea una mascota y verifica que GET /pet/{petId} la obtiene con esquema válido (200)."""
    payload = _generar_payload_mascota()
    resp_creacion = api.post("/pet", json=payload)
    assert resp_creacion.status_code == 200
    pet_id = resp_creacion.json().get("id", payload["id"])

    response = api.get(f"/pet/{petId}".replace("{petId}", str(pet_id)))

    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=PET_SCHEMA)
    assert data["id"] == pet_id


def test_get_pet_por_id_no_existente_404(api):
    """Verifica 404 al consultar un ID de mascota que no existe."""
    id_inexistente = 999999999998
    response = api.get(f"/pet/{id_inexistente}")

    assert response.status_code == 404


def test_get_pet_por_id_formato_invalido_400(api):
    """Verifica 400 cuando el parámetro petId no es un entero (ID inválido)."""
    response = api.get("/pet/cadena_invalida_no_int")

    assert response.status_code == 400

def test_post_pet_con_formulario_exitoso_200(api):
    """Verifica que POST /pet/{petId} actualiza name y status vía query params (200)."""
    payload = _generar_payload_mascota(status="available")
    resp_creacion = api.post("/pet", json=payload)
    assert resp_creacion.status_code == 200
    pet_id = resp_creacion.json().get("id", payload["id"])

    nuevo_nombre = f"form-{uuid.uuid4().hex[:6]}"
    nuevo_estado = "sold"

    response = api.post(f"/pet/{pet_id}", params={"name": nuevo_nombre, "status": nuevo_estado})

    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=PET_SCHEMA)
    assert data["name"] == nuevo_nombre
    assert data["status"] == nuevo_estado


def test_post_pet_con_formulario_id_invalido_400(api):
    """Verifica 400 al invocar POST /pet/{petId} con un ID inválido (no entero)."""
    response = api.post("/pet/id_no_valido", params={"name": "test", "status": "pending"})

    assert response.status_code in (400, 404)

@pytest.mark.parametrize("status_val", ["available", "pending", "sold"])
def test_get_pet_buscar_por_status_valido_200(api, status_val):
    """Verifica que GET /pet/findByStatus responde 200 y valida la lista con cada estado permitido."""
    # Sembrar una mascota con ese estado para asegurar datos
    payload = _generar_payload_mascota(status=status_val)
    api.post("/pet", json=payload)

    response = api.get("/pet/findByStatus", params={"status": status_val})

    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=PET_LIST_SCHEMA)
    # Si devuelve elementos, todos deben tener el estado filtrado
    for pet in data:
        if "status" in pet:
            assert pet["status"] == status_val


def test_get_pet_buscar_por_status_invalido_400(api):
    """Verifica 400 al buscar mascotas con un estado que no pertenece al enum."""
    response = api.get("/pet/findByStatus", params={"status": "estado_completamente_invalido"})

    assert response.status_code == 400

# Pruebas GET /pet/findByTags (findPetsByTags)
# ===========================================================================

def test_get_pet_buscar_por_tags_exitoso_200(api):
    """Crea una mascota con un tag específico y verifica que GET /pet/findByTags la encuentra (200)."""
    tag_unico = f"tag-{uuid.uuid4().hex[:8]}"
    payload = _generar_payload_mascota(tag_name=tag_unico)
    resp_creacion = api.post("/pet", json=payload)
    assert resp_creacion.status_code == 200

    response = api.get("/pet/findByTags", params={"tags": tag_unico})

    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=PET_LIST_SCHEMA)
    nombres_tags = [
        tag["name"]
        for pet in data
        for tag in pet.get("tags", [])
        if "name" in tag
    ]
    assert tag_unico in nombres_tags


def test_get_pet_buscar_por_tags_multiples_200(api):
    """Verifica búsqueda por múltiples tags separados por coma."""
    tag1 = f"tag1-{uuid.uuid4().hex[:6]}"
    tag2 = f"tag2-{uuid.uuid4().hex[:6]}"

    payload1 = _generar_payload_mascota(tag_name=tag1)
    payload2 = _generar_payload_mascota(tag_name=tag2)
    api.post("/pet", json=payload1)
    api.post("/pet", json=payload2)

    response = api.get("/pet/findByTags", params={"tags": f"{tag1},{tag2}"})

    assert response.status_code == 200
    data = response.json()
    validate(instance=data, schema=PET_LIST_SCHEMA)

def test_delete_order_no_encontrada_404(api):
    """Verifica 404 al intentar eliminar una orden inexistente (ID válido < 1000)."""
    # Rango < 1000 según la descripción del contrato
    order_id = random.randint(900, 999)
    # Se asegura que la orden no exista eliminándola primero o consultando una ficticia
    response = api.delete(f"/store/order/{order_id}")

    # Puede retornar 404 si no existe o 200 si existía previamente
    assert response.status_code in (200, 404)
    if response.status_code == 200:
        # Segundo intento consecutivo debe confirmar la no existencia (404)
        segundo_intento = api.delete(f"/store/order/{order_id}")
        assert segundo_intento.status_code == 404

    """Verifica error (400 o 404) para IDs superiores a 1000 según especificación del recurso."""

    response = api.delete(f"/store/order/{order_id_grande}")
