# JSON Schemas from OpenAPI Contract
USER_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["email", "id"],
    "properties": {
        "email": {"type": "string"},
        "is_active": {"type": "boolean"},
        "is_superuser": {"type": "boolean"},
        "full_name": {"type": ["string", "null"]},
        "id": {"type": "string"},
        "created_at": {"type": ["string", "null"]},
    },
}

USERS_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["data", "count"],
    "properties": {
        "data": {"type": "array", "items": USER_PUBLIC_SCHEMA},
        "count": {"type": "integer"},
    },
}

ITEM_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["title", "id", "owner_id"],
    "properties": {
        "title": {"type": "string"},
        "description": {"type": ["string", "null"]},
        "id": {"type": "string"},
        "owner_id": {"type": "string"},
        "created_at": {"type": ["string", "null"]},
    },
}

ITEMS_PUBLIC_SCHEMA = {
    "type": "object",
    "required": ["data", "count"],
    "properties": {
        "data": {"type": "array", "items": ITEM_PUBLIC_SCHEMA},
        "count": {"type": "integer"},
    },
}

TOKEN_SCHEMA = {
    "type": "object",
    "required": ["access_token"],
    "properties": {
        "access_token": {"type": "string"},
        "token_type": {"type": "string"},
    },
}

MESSAGE_SCHEMA = {
    "type": "object",
    "required": ["message"],
    "properties": {"message": {"type": "string"}},
}

HTTP_VALIDATION_ERROR_SCHEMA = {
    "type": "object",
    "required": ["detail"],
    "properties": {
        "detail": {
            "type": "array",
            "items": {
                "type": "object",
                "required": ["loc", "msg", "type"],
                "properties": {
                    "loc": {
                        "type": "array",
                        "items": {"type": ["string", "integer"]},
                    },
                    "msg": {"type": "string"},
                    "type": {"type": "string"},
                },
            },
        }
    },
}

def test_delete_users_me_for_dedicated_user(base_url):
    # To avoid deleting the admin user, create a dedicated user and use their own session
    user_email = random_email()
    user_password = "userPassword123!"

    reg_res = requests.post(
        f"{base_url}/api/v1/users/signup",
        json={"email": user_email, "password": user_password},
    )
    assert reg_res.status_code == 200

    token_res = requests.post(
        f"{base_url}/api/v1/login/access-token",
        data={"username": user_email, "password": user_password},
    )
    assert token_res.status_code == 200
    token = token_res.json()["access_token"]

    user_session = requests.Session()
    user_session.headers.update({"Authorization": f"Bearer {token}"})

    del_res = user_session.delete(f"{base_url}/api/v1/users/me")
    assert del_res.status_code == 200
    data = del_res.json()
    validate(instance=data, schema=MESSAGE_SCHEMA)
