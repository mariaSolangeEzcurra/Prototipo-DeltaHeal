pet_schema = {
    "type": "object",

order_schema = {

        "complete": {"type": "boolean"},
    },
    "additionalProperties": False,
}

user_schema = {

def validate_response_schema(response_json, schema):
    """Helper to validate a JSON response against a given schema."""
    try:
        validate(instance=response_json, schema=schema)
    except ValidationError as e:
        pytest.fail(f"Schema validation failed: {e.message} for data: {response_json}")

def generate_random_string(length=10):
    return "".join(random.choice(string.ascii_lowercase) for _ in range(length))


def generate_random_id():
    return random.randint(100000, 999999)


def generate_pet_data(pet_id=None, status="available"):
    return {
        "id": pet_id if pet_id is not None else generate_random_id(),
        "name": f"doggie-{generate_random_string(5)}",
        "category": {"id": generate_random_id(), "name": "Dogs"},
        "photoUrls": [f"http://example.com/photo-{generate_random_string(5)}.jpg"],
        "tags": [{"id": generate_random_id(), "name": f"tag-{generate_random_string(3)}"}],
        "status": status,
    }


def generate_user_data(username=None):
    user_id = generate_random_id()
    return {
        "id": user_id,
        "username": username if username else f"user-{generate_random_string(5)}",
        "firstName": f"John-{generate_random_string(3)}",
        "lastName": f"Doe-{generate_random_string(3)}",
        "email": f"john.doe.{generate_random_string(3)}@example.com",
        "password": "password123",
        "phone": "123-456-7890",
        "userStatus": 1,
    }


def generate_order_data(order_id=None, pet_id=None, status="placed"):

def test_post_pet_update_with_form_200(api, create_pet):
    pet = create_pet()
    new_name = f"form-updated-{generate_random_string(5)}"
    new_status = "pending"
    response = api.post(f"/pet/{pet['id']}", params={"name": new_name, "status": new_status})
    assert response.status_code == 200
    # The API returns the updated pet, but the schema is for ApiResponse.
    # The OpenAPI spec for this endpoint's 200 response content is Pet schema,
    # but the actual Petstore API returns ApiResponse.
    # We'll validate against ApiResponse as it's more common for form updates to return a status.
    # If the API actually returned Pet, this would need adjustment.
    response_json = response.json()
    validate_response_schema(response_json, api_response_schema)
    assert response_json["message"] == str(pet["id"])

    # Verify update by fetching the pet
    get_response = api.get(f"/pet/{pet['id']}")
    assert get_response.status_code == 200
    updated_pet = get_response.json()
    assert updated_pet["name"] == new_name
    assert updated_pet["status"] == new_status
