def validate_response_schema(response_json, schema):
    """Helper to validate a JSON response against a given schema."""
    try:
        validate(instance=response_json, schema=schema)
    except ValidationError as e:
        pytest.fail(f"Schema validation failed: {e.message} for data: {response_json}")

def generate_random_string(length=10):
    return "".join(random.choice(string.ascii_lowercase) for _ in range(length))

def generate_random_id():
    return random.randint(100000, 999999)

def generate_pet_data(pet_id=None, status="available"):
    return {
        "id": pet_id if pet_id is not None else generate_random_id(),
        "name": f"doggie-{generate_random_string(5)}",
        "category": {"id": generate_random_id(), "name": "Dogs"},
        "photoUrls": [f"http://example.com/photo-{generate_random_string(5)}.jpg"],
        "tags": [{"id": generate_random_id(), "name": f"tag-{generate_random_string(3)}"}],
        "status": status,
    }

def generate_user_data(username=None):
    user_id = generate_random_id()
    return {
        "id": user_id,
        "username": username if username else f"user-{generate_random_string(5)}",
        "firstName": f"John-{generate_random_string(3)}",
        "lastName": f"Doe-{generate_random_string(3)}",
        "email": f"john.doe.{generate_random_string(3)}@example.com",
        "password": "password123",
        "phone": "123-456-7890",
        "userStatus": 1,
    }

def generate_order_data(order_id=None, pet_id=None, status="placed"):
    return {
        "id": order_id if order_id is not None else generate_random_id(),
        "petId": pet_id if pet_id is not None else generate_random_id(),
        "quantity": random.randint(1, 10),
        "shipDate": (datetime.now() + timedelta(days=1)).isoformat(timespec="seconds") + "Z",
        "status": status,
        "complete": False,
    }

@pytest.fixture
def create_pet(api):
    """Fixture to create a pet and return its ID."""

    def _create_pet(status="available"):
        pet_data = generate_pet_data(status=status)
        response = api.post("/pet", json=pet_data)
        assert response.status_code == 200
        created_pet = response.json()
        validate_response_schema(created_pet, pet_schema)
        return created_pet

    yield _create_pet

@pytest.fixture

        if pet_id is None:

    yield _create_order
